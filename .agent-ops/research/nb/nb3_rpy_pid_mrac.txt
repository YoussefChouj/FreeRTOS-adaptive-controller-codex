# %% [CODE cell 2]
# PID Nominal Controller — LQR-Tuned (WITH INTEGRAL ACTION)
# ============================================

# ── 1. Linear Plant Model (accel-domain) ────────────────────────────────────
# x = [θ, ω]T
A_p  = np.array([[0., 1.],
                 [0., 0.]])
B_p  = np.array([[0.],
                 [1.]])
C_p  = np.array([[1., 0.]])   # Output = angle

# To tune PID directly with python-control's LQR, we augment the plant with an integral state.
# state vector: X = [θ, ω, z_int]T
# z_int_dot = θ (angle error integration assuming θ_ref = 0 for LQR design)
A_aug = np.block([
    [A_p, np.zeros((2, 1))],
    [C_p, 0]
])
B_aug = np.vstack((B_p, 0))

# ── 2. LQR Tuning (Penalty Matrices) ─────────────────────────────────────────

# Weights: Q = diag([w_theta, w_omega, w_integral]), R = w_control
Q_pr = np.diag([80.0,   # Proportional penalty
                1.0,     # Derivative penalty
                8.0])  # Integral penalty (higher = faster elimination of SS error)
R_pr = np.array([[0.5]])

Q_yaw = np.diag([80.0,
                 1.0,
                 8.0])
R_yaw = np.array([[0.5]])

# ── 3. LQR Optimization to get K_pid = [Kp, Kd, Ki] ──────────────────────────
# lqr() returns K corresponding to u = -K * X
K_aug_pitch_roll, _, _ = ct.lqr(A_aug, B_aug, Q_pr, R_pr)
K_aug_yaw, _, _   = ct.lqr(A_aug, B_aug, Q_yaw, R_yaw)

# Extract individual normalized gains
K1_pitch, K1_roll = K_aug_pitch_roll[0, 0] , K_aug_pitch_roll[0, 0]  # Kp
K3_pitch, K3_roll = K_aug_pitch_roll[0, 1], K_aug_pitch_roll[0, 1]  # Kd
K2_pitch, K2_roll = K_aug_pitch_roll[0, 2] , K_aug_pitch_roll[0, 2]  # Ki (on integral state z)

K1_yaw = K_aug_yaw[0, 0]
K3_yaw = K_aug_yaw[0, 1]
K2_yaw = K_aug_yaw[0, 2]

# Rate filter for dirty derivative
LAMBDA = 50.0

# ── 4. Build Augmented Reference Model for MRAC ──────────────────────────────
def build_augmented_model(K1, K2, K3):
    """
    Construct augmented 3rd-order reference model for one axis.
    States: ξ_r = [θ_r, ω_r, z_r]^T  (where z_r is integral of tracking error)
    
    Using the LQR-derived PID gains.
    """
    F     = np.eye(2) + B_p * K3 @ C_p
    F_inv = np.linalg.inv(F)
    G     = A_p - B_p * K1 @ C_p

    Ar = np.block([
        [F_inv @ G,   -(F_inv @ B_p) * K2],
        [C_p,          np.zeros((1,1))    ]
    ])
    
    Br = np.vstack([F_inv @ B_p * K1,
                    [[-1.]]])
    
    Ba = np.vstack([F_inv @ B_p,
                    [[0.]]])

    return F, F_inv, G, Ar, Br, Ba

(F_pr, F_inv_pr, G_pr, Ar_pr, Br_pr, Ba_pr) = build_augmented_model(K1_pitch, K2_pitch, K3_pitch)
(F_yaw, F_inv_yaw, G_yaw, Ar_yaw, Br_yaw, Ba_yaw) = build_augmented_model(K1_yaw, K2_yaw, K3_yaw)

# ── 5. Solving Lyapunov Equation for MRAC Weight Update ──────────────────────
P_pr  = ct.lyap(Ar_pr.T, np.eye(3))
P_yaw = ct.lyap(Ar_yaw.T, np.eye(3))

# Check stability (sanity check)
eigs_ar_pr  = np.linalg.eigvals(Ar_pr)
eigs_ar_yaw = np.linalg.eigvals(Ar_yaw)

print(f"  Real torque gains : Kp={Jy*K1_pitch:.5f}, Kd={Jy*K3_pitch:.5f}, Ki={Jy*K2_pitch:.5f}")
print(f"  Ar eigenvalues    : {[f'{e:.2f}' for e in eigs_ar_pr]}")

# PID Nominal Controller — Pole-Placement Tuned (WITH INTEGRAL ACTION)
# ============================================

import control as ct

# ── 1. Linear Plant Model (accel-domain) ────────────────────────────────────
# x = [θ, ω]T
A_p  = np.array([[0., 1.],
                 [0., 0.]])
B_p  = np.array([[0.],
                 [1.]])
C_p  = np.array([[1., 0.]])   # Output = angle

# To tune PID directly, we augment the plant with an integral state.
# state vector: X = [θ, ω, z_int]T
# z_int_dot = θ (angle error integration assuming θ_ref = 0 for pole design)
A_aug = np.block([
    [A_p, np.zeros((2, 1))],
    [C_p, 0]
])
B_aug = np.vstack((B_p, 0))

# ── 2. POLE PLACEMENT DESIGN ────────────────────────────────────────────────
# For a 3rd order system (Angle, Rate, Integral), we place 3 poles.
# Target: Fast response with moderate damping to avoid excessive oscillations.
# Pitch/Roll Poles
omega_n_pr = 10.0              # Adjusted from 8.0: Safer for 40ms motor lag
zeta_pr    = 0.8              # Increased damping (0.707 -> 0.8) for stability
p_pr_fast  = -1.0            # Slower integral pole to improve phase margin
poles_pr = np.array([
    -zeta_pr*omega_n_pr + 1j*omega_n_pr*np.sqrt(1-zeta_pr**2),
    -zeta_pr*omega_n_pr - 1j*omega_n_pr*np.sqrt(1-zeta_pr**2),
    p_pr_fast
])

# Yaw Poles (Usually slower due to lower control authority)
omega_n_yaw = 5.0             # Adjusted from 4.0
zeta_yaw    = 0.8             
p_yaw_fast  = -3.0            
poles_yaw = np.array([
    -zeta_yaw*omega_n_yaw + 1j*omega_n_yaw*np.sqrt(1-zeta_yaw**2),
    -zeta_yaw*omega_n_yaw - 1j*omega_n_yaw*np.sqrt(1-zeta_yaw**2),
    p_yaw_fast
])

# Calculate Gains using Ackermann's formula or Place
K_aug_pitch_roll = ct.place(A_aug, B_aug, poles_pr)
K_aug_yaw        = ct.place(A_aug, B_aug, poles_yaw)

# Extract individual normalized gains: K = [Kp, Kd, Ki]
# Note: In our augmentation [θ, ω, z], the gains map as:
# u = -K1*θ - K3*ω - K2*z
K1_pitch = K_aug_pitch_roll[0, 0] # Kp
K3_pitch = K_aug_pitch_roll[0, 1] # Kd
K2_pitch = K_aug_pitch_roll[0, 2] # Ki

K1_yaw = K_aug_yaw[0, 0] # Kp
K3_yaw = K_aug_yaw[0, 1] # Kd
K2_yaw = K_aug_yaw[0, 2] # Ki

# Assign Roll (same as pitch)
K1_roll, K2_roll, K3_roll = K1_pitch, K2_pitch, K3_pitch

# Rate filter for dirty derivative
LAMBDA = 50.0

# ── 3. Build Augmented Reference Model for MRAC ──────────────────────────────
def build_augmented_model(K1, K2, K3):
    """
    Construct augmented 3rd-order reference model for one axis.
    States: ξ_r = [θ_r, ω_r, z_r]^T  (where z_r is integral of tracking error)
    """
    # System: ξ_dot = (A - B*K)ξ + B*Kp*r_filtered 
    # But since we have integral action, B_ref enters at the error junction.
    
    # Simple closed-loop ref model
    Ar = A_aug - B_aug @ np.array([[K1, K3, K2]])
    
    # Reference input scaling (so steady state angle matches reference)
    # At steady state x_dot = 0 -> Ar*x + Br*r = 0 -> x = -inv(Ar)*Br*r
    # We want x[0] = r.
    Br = np.array([[0.], [0.], [-1.0]]) # Reference enters integral state directly

    return Ar, Br

(Ar_pr, Br_pr) = build_augmented_model(K1_pitch, K2_pitch, K3_pitch)
(Ar_yaw, Br_yaw) = build_augmented_model(K1_yaw, K2_yaw, K3_yaw)

# ── 4. Solving Lyapunov Equation for MRAC ───────────────────────────────────
# Solve A.T*P + P*A = -Q
P_pr  = ct.lyap(Ar_pr.T, np.eye(3))
P_yaw = ct.lyap(Ar_yaw.T, np.eye(3))

# Check stability (sanity check)

# ── 5. PERFORMANCE METRICS ──────────────────────────────────────────────────
def get_metrics(Ar):
    # Step response of reference model to check rise time and overshoot
    sys = ct.ss(Ar, [[0.],[0.],[-1.0]], [1, 0, 0], 0)
    t, y = ct.step_response(sys, T=np.linspace(0, 2, 500))
    
    rise_time = t[np.where(y >= 0.9)[0][0]] if any(y >= 0.9) else np.nan
    overshoot = (np.max(y) - 1.0) * 100 if np.max(y) > 1.0 else 0.0
    settling_time = t[np.where(np.abs(y-1.0) > 0.02)[0][-1]] if any(np.abs(y-1.0) > 0.02) else 0.0
    
    return rise_time, overshoot, settling_time

rt_pr, os_pr, st_pr = get_metrics(Ar_pr)
rt_yaw, os_yaw, st_yaw = get_metrics(Ar_yaw)

# PID Step Response Verification (all 3 axes)
# ============================================
import scipy.signal as sig

theta_ref_deg = 20.0          # commanded step [deg]
theta_ref_rad = np.radians(theta_ref_deg)

# ── 3-state augmented closed-loop: x = [θ, ω, z_int] ────────────────────────
# A_cl = A_aug - B_aug @ K_aug  (full PID feedback)
# B_cl = Br_pr = [[0], [K1], [-1]] (reference enters: feedforward Kp + integral)
#
# NOTE: Motor lag (tau_m = 60ms) is added as an extra state to reflect real hardware.
#       Without it, the step response is too optimistic.

# LQR gains — use K_aug_pitch_roll (set in cell 3)
K_pitch_norm = K_aug_pitch_roll   # 1×3: [Kp, Kd, Ki] pitch & roll (Jx=Jy)
K_yaw_norm   = K_aug_yaw          # 1×3: [Kp, Kd, Ki] yaw

# Real gain matrices for saturation check (multiply by inertia)
K_pitch_real = Jy * K_aug_pitch_roll   # Nm / rad
K_yaw_real   = Jz * K_aug_yaw

def step_response_axis(K_norm, Br, J, label, color, ax, ax_torque, torque_limit):
    """Closed-loop + motor-lag step response.
    States: [θ, ω, z_int, u_motor]  (4th state = actual motor torque)
    Motor: u_dot = (1/tau_m)*(u_cmd - u_motor)   → adds lag.
    """
    # Base closed-loop (accel domain, no motor lag)
    A_cl  = A_aug - B_aug @ K_norm          # 3×3
    B_cl  = Br                               # 3×1  (from reference model build)

    # Augment with motor first-order lag: new state u_m  (index 3)
    tau = MOTOR_TAU
    n = A_cl.shape[0]
    A4 = np.block([
        [A_cl,              -B_aug          ],
        [np.zeros((1, n)),  np.array([[-1.0/tau]])]
    ])
    B4 = np.vstack([B_cl, [[0.]]])
    
    # Outputs: [Angle, Motor Torque command (Nm)]
    # u_cmd = -K_norm @ x + Br_const * theta_ref
    # Motor torque command (Nm) = J * u_cmd
    C4 = np.array([
        [1., 0., 0., 0.],                            # θ
        [-J*K_norm[0,0], -J*K_norm[0,1], -J*K_norm[0,2], 0.] # τ_cmd (Nm)
    ])
    # D component: τ_cmd also depends on reference directly through B_cl
    # If Br is from Ar build, it matches reference scaling.
    D4 = np.array([[0.], [J * B4[0,0]]]) 

    sys_cl = sig.StateSpace(A4, B4, C4, D4)
    t = np.linspace(0, 3.0, 800)
    _, response = sig.step(sys_cl, T=t)
    
    y_deg = response[:,0] * np.degrees(1.0) * theta_ref_rad
    u_torque = response[:,1] * theta_ref_rad

    ax.plot(t, y_deg, color=color, lw=2, label=f'{label} Angle')
    ax.axhline(theta_ref_deg, color='k', lw=0.8, ls=':', alpha=0.6)
    
    ax_torque.plot(t, u_torque, color=color, lw=2, label=f'{label} Torque')
    ax_torque.axhline(torque_limit, color='r', ls='--', alpha=0.5, label='Limit')
    ax_torque.axhline(-torque_limit, color='r', ls='--', alpha=0.5)
    
    # Mark 90% rise time
    try:
        rise_idx = np.where(y_deg >= 0.90 * theta_ref_deg)[0][0]
        ax.axvline(t[rise_idx], color=color, lw=0.8, ls='--', alpha=0.5,
                   label=f't_rise={t[rise_idx]:.2f}s')
    except IndexError: pass

    return np.max(np.abs(u_torque))

## PID-MRAC for Drone Attitude Control — Mapping from Mass-Spring-Damper

### Structural Analogy

| Mass-Spring-Damper | Drone Axis (e.g. Pitch) |
|---|---|
| position $x$ | angle $\theta$ |
| velocity $\dot{x}$ | angular rate $\omega$ |
| mass $m$, $\Lambda = 1/m$ | inertia $J_y$, $\Lambda = 1/J_y$ |
| spring $\alpha x$ + damper $\beta \dot{x}$ | aero drag $C_d \omega$ + gyro coupling $C_g \theta$ |
| $\mathbf{W}_0 = [-\alpha, -\beta]^\top$ | $\mathbf{W}_0 = [-C_g, -C_d]^\top$ (true uncertainty) |
| plant input $u$ | commanded torque $\tau$ [Nm] |

### Architecture

$$\tau = \underbrace{-K_1(\theta - \theta_{ref}) - K_2 z - K_3 \hat{\omega}}_{\tau_{nom}\ \text{(LQR-PID)}} + \underbrace{-\hat{\mathbf{W}}^\top \boldsymbol{\phi}}_{\tau_{ada}\ \text{(MRAC adaptive)}}$$

- $z = \int (\theta - \theta_{ref})\,dt$ — integral state  
- $\hat{\omega}$ — filtered rate via dirty derivative ($\lambda$-filter)  
- $\boldsymbol{\phi} = [\theta,\ \omega,\ \tau_{nom}]^\top$ — regressor (3-element)  
- $\hat{\mathbf{W}}$ — online estimated weights (3-element)

### Augmented Reference Model (3rd order — angle, rate, integral)

$$\dot{\xi}_r = A_r \xi_r + B_r \theta_{ref}, \quad \xi_r = [\theta_r, \omega_r, z_r]^\top$$

$$A_r = \begin{bmatrix} F^{-1}G & -F^{-1}B K_2 \\ E & 0 \end{bmatrix}, \quad B_r = \begin{bmatrix} F^{-1}B K_1 \\ -1 \end{bmatrix}$$

where $F = I + B K_3 E$, $G = A - B K_1 E$ encodes the PID structure into the reference model poles.

### Lyapunov Weight Update

$$\dot{\hat{\mathbf{W}}} = \gamma \, \boldsymbol{\phi} \, (\xi - \xi_r)^\top P \, B_a, \qquad A_r^\top P + P A_r = -I$$

where $B_a = [F^{-1}B;\ 0]$ is the augmented input direction.

### LQR Gain Mapping (from cell 3)
- $K_1 = K_{p,norm}$ (proportional) — from LQR on $Q_{pr}, R_{pr}$  
- $K_3 = K_{d,norm}$ (derivative) — from LQR on $Q_{pr}, R_{pr}$  
- $K_2$ (integral) — designed separately to maintain augmented Hurwitz stability

# ── Simulation Settings ───────────────────────────────────────────────────────
dt_sim = 0.005       # time step [s]
ft_sim = 40.0        # total time [s]

# ── True Uncertainty (W0) ─────────────────────────────────────────────────────
# Physical: aero drag proportional to ω, gyro coupling proportional to θ
# In normalised accel units divide by Jy so plant eqn is:
#   θ̈ = u + W0ᵀ·[θ, ω]   (pure accel, no Jy in the loop)
C_gyro_true = 10.8    # [1/s²]  gyroscopic coupling (angle → accel disturbance)
C_drag_true = 10.5    # [1/s]   aerodynamic drag    (rate  → accel disturbance)
W0 = np.array([[-C_gyro_true],
               [-C_drag_true]])    # (2×1) true uncertainty in accel units

Lambda = 1.0   # accel-domain plant gain (≡1 since we normalised by Jy already)

# ── Controller Parameters ─────────────────────────────────────────────────────
K1 = K1_pitch    # proportional  (LQR, accel units)
K2 = K2_pitch              # integral
K3 = K3_pitch    # derivative    (LQR, accel units)
lam = LAMBDA               # dirty-derivative filter pole [rad/s]

# Augmented reference model (from cell 3)
Ar  = Ar_pr;  Br  = Br_pr;  Ba  = Ba_pr;  P = P_pr

# Adaptation gain — tune: higher → faster learning, lower → less noise sensitivity
gam = 10

# ── Initial Conditions ────────────────────────────────────────────────────────
x     = np.array([[0.], [0.]])  # [θ, ω]
z     = np.array([[0.]])        # integral of tracking error
q     = np.array([[0.]])        # λ-filter state (approx filtered angle)
xir   = np.zeros((3, 1))        # augmented reference model state
W_hat = np.zeros((3, 1))        # weight estimates [w_θ, w_ω, w_τnom]

    if   t <= 10:  return  np.radians(10.)
    elif t <= 20:  return  np.radians(-10.)
    elif t <= 30:  return  np.radians(15.)
    else:          return  np.radians(-5.)

def reference_time_varying(t, Amp, Freq=0.1):

    return np.radians(Amp) * np.sin(2 * np.pi * Freq * t)

    c = reference_time_varying(t, Freq=2, Amp=10.0)     # time-varying reference (sine wave)
    # Measured output (angle)
    y = C_p @ x                                     # (1×1)

    # λ-filter: low-pass angle to get dirty derivative (rate estimate)
    q   = q   + dt_sim * (-lam * (q - y))
    yd  = -lam * (q - y)                          # ≈ ω (filtered derivative)

    # ── Nominal PID ──────────────────────────────────────────────────────────
    err        = (y - c).item()
    tau_nom    = -K1 * err - K2 * z.item() - K3 * yd.item()   # scalar
    z          = z + dt_sim * (y - c)             # integral update

    # ── Regressor & Adaptive Control ─────────────────────────────────────────
    phi = np.array([[x[0,0]],                     # θ
                    [x[1,0]],                     # ω
                    [tau_nom]])                    # τ_nom  (3×1)

    tau_ada = -(W_hat.T @ phi).item()             # scalar

    # ── Total control (normalised accel units) ───────────────────────────────
    u = tau_nom + tau_ada                         # scalar

    # ── Augmented state & reference model ────────────────────────────────────
    xi   = np.vstack([x, z])                      # (3×1)
    xir  = xir + dt_sim * (Ar @ xir + Br * c)    # reference model
    e    = xi - xir                               # augmented tracking error

    # ── Weight Update (Lyapunov) ─────────────────────────────────────────────
    W_hat = W_hat + dt_sim * (gam * phi @ (e.T @ P @ Ba))

    # ── True Plant (with uncertainty W0) ─────────────────────────────────────
    # θ̈ = u + W0ᵀ·[θ, ω]
    phi_unc = x                                   # (2×1)
    x_dot   = A_p @ x + B_p * Lambda * (u + (W0.T @ phi_unc).item())
    x       = x + dt_sim * x_dot

    # ── Record ───────────────────────────────────────────────────────────────
    t_rec.append(t)
    x_rec.append(x.flatten())
    xir_rec.append(xir.flatten())
    u_rec.append(u)
    tau_nom_rec.append(tau_nom)
    tau_ada_rec.append(tau_ada)
    w_rec.append(W_hat.flatten())
    c_rec.append(c)
    z_rec.append(z.item())

## Cascaded PID-MRAC Architecture
Industry standard flight controllers (Betaflight, PX4) use a cascaded loop. 
Instead of tuning the LQR for a single loop, we can map our optimal LQR state-space gains directly into a Cascaded architecture. 

Here is the algebraic mapping:
1. **Inner Rate Loop:** Acts as the system damping. $P_{in} = K_d$
2. **Outer Angle Loop:** Acts as the system stiffness. $P_{out} = K_p / K_d$, $I_{out} = K_i / K_d$

**The Massive Advantage:**
1. We can now limit the maximum rotational rate (`clip(omega_cmd)`) so large stick inputs don't rip the drone apart.
2. We only need a **1st-order MRAC** applied to the inner loop. The outer loop is purely kinematic and requires no adaptation.

# %% [MD cell 11]
## Beginner's Guide: How these PID Gains are Calculated

If you are new to control theory, "LQR" and "Augmented States" can sound like magic. Here is the plain-English explanation of the logic we used in Cell 3 and Cell 4.

### 1. The Goal: Automatic Tuning
Normally, you tune PID gains by hand (Betaflight style), trial and error. 
**LQR (Linear Quadratic Regulator)** is a math tool that does the tuning for you. You tell it:
* "I care this much about **Angle Error**."
* "I care this much about **Battery/Motor Stress**."
* LQR then calculates the **mathematically perfect** balance to get the fastest response without overworking the motors.

### 2. Why "Augmented" States?
A basic drone model only knows about **Position** (Angle) and **Velocity** (Rate). 
But a real PID controller needs an **Integral (I)** term to fix gravity or wind offsets. 
To let the LQR math "see" the Integral term, we create a fake 3rd state called the **Integral State**. This is called "Augmenting" the system. Now LQR can calculate a gain for all three: $K_p$, $K_d$, and $K_i$.

### 3. Normalized vs. Real Gains
In this notebook, we do math in the "Acceleration Domain." 
* **Normalized Gain:** Tells us how much *acceleration* ($rad/s^2$) to command for 1 degree of error. This is independent of the drone's weight.
* **Real Gain (Nm):** We multiply the normalized gain by the **Inertia (J)**. This tells the actual motor how many **Newton-meters** of torque to apply.

### 4. The "Reality Check" (Motor Lag)
In Cell 4, we added a **60ms Motor Lag**. 
In a perfect simulation, motors react instantly. In real life, propellers take time to spin up. By adding this lag to our simulation, we can see if our PID gains are "too sharp." If the graph starts oscillating, we know our PID gains are too high for your specific 5-inch props and motors.

**Summary for you:**
* **Q Matrix:** Your "Priority List" (Higher numbers = Snappier drone).
* **R Matrix:** Your "Efficiency List" (Higher numbers = Smoother/Slower drone).
* **Step Plot:** Shows you exactly how your drone will tilt when you move the stick.

## Professional Sequential LQR Tuning (Successive Loop Closure)
Instead of algebraically unpacking a single LQR, industry standard controllers tune two completely independent LQR systems.

### 1. Inner Loop LQR (Rate Controller -> Full PID)
To get LQR to output a full PID ($K_{p\_in}$, $K_{i\_in}$, $K_{d\_in}$), we must teach it about the **Motor Time Constant ($\tau_m$)**. By augmenting the plant with motor latency $\dot{\tau}_{real} = \frac{1}{\tau_m}(\tau_{cmd} - \tau_{real})$, LQR will produce a predictive $K_d$ term on the angular acceleration ($\dot{\omega}$) to combat the motor delay!
* State: $[\omega,\ \dot{\omega},\ \int \omega]^T$

### 2. Outer Loop LQR (Angle Controller -> PI)
Once the inner loop is tuned, the outer loop only handles kinematics ($\dot{\theta} = \omega_{cmd}$). Because this relies on pure geometry instead of mass or motors, LQR naturally outputs a simple PI controller ($K_{p\_out}$, $K_{i\_out}$) and calculates that $K_{d\_out}$ is unnecessary (since rate is already handled inside).
* State: $[\theta,\ \int \theta]^T$

def compute_step_specs(t, y, r, settle_band=0.02, eps=1e-9):
    """Compute rise time, settling time, and percent overshoot for each reference step."""
    step_idx = np.where(np.abs(np.diff(r)) > eps)[0] + 1
    if len(step_idx) == 0:
        return []

    segment_starts = step_idx
    segment_ends = list(step_idx[1:]) + [len(r) - 1]
    specs = []

    for i, (k0, k1) in enumerate(zip(segment_starts, segment_ends), start=1):
        if k0 == 0 or k1 <= k0:
            continue

        r_prev = r[k0 - 1]
        r_new = r[k0]
        amp = r_new - r_prev
        if abs(amp) < eps:
            continue

        t_seg = t[k0:k1 + 1]
        y_seg = y[k0:k1 + 1]

        # Normalize so positive/negative steps are handled identically
        z = (y_seg - r_prev) / amp

        # Rise time (10% -> 90%)
        idx10 = np.where(z >= 0.10)[0]
        idx90 = np.where(z >= 0.90)[0]
        if len(idx10) > 0 and len(idx90) > 0 and idx90[0] >= idx10[0]:
            rise_time = float(t_seg[idx90[0]] - t_seg[idx10[0]])
        else:
            rise_time = np.nan

        # Percent overshoot
        overshoot_pct = max(0.0, float((np.max(z) - 1.0) * 100.0))

        # Settling time (first time after which response stays within ±2% band)
        tol = settle_band * abs(amp)
        err_abs = np.abs(y_seg - r_new)
        settling_time = np.nan
        for j in range(len(err_abs)):
            if np.all(err_abs[j:] <= tol):
                settling_time = float(t_seg[j] - t_seg[0])
                break

        specs.append({
            "step": i,
            "t_step": float(t[k0]),
            "from_deg": float(np.degrees(r_prev)),
            "to_deg": float(np.degrees(r_new)),
            "rise_s": rise_time,
            "settle_s": settling_time,
            "overshoot_pct": overshoot_pct,
        })

    return specs

    # Inner MRAC
    omega_r += dt_sim * (-Kp_in_kin * omega_r + Kp_in_kin * omega_cmd)
    err_mrac = omega - omega_r
    
    phi = np.array([[1.0], [theta], [omega]]) # Bias, Angle, Rate
    
    if adaptation:
        tau_ada = -(W_hat.T @ phi).item()
        W_dot = Gamma @ phi * err_mrac
        W_hat += dt_sim * W_dot
        # Simple Projection (Clipping)
        W_hat = W_hat
    else:
        tau_ada = 0.0
        
    tau_total = tau_nom + tau_ada
    
    # ── ACTUATOR MIXER (Nm -> PWM^2) ─────────────────────────
    # We simulate a single axis (e.g., Pitch). 
    # M2 and M3 go UP (Hover + Torque), M1 and M4 go DOWN (Hover - Torque)
    pwm2_delta = tau_total * TORQUE_TO_PWM2
    
    motor_front_pwm2 = HOV_THRUST_SQ + pwm2_delta
    motor_rear_pwm2  = HOV_THRUST_SQ - pwm2_delta
    
    # Hardware Saturation (Clipping to 0 and Max PWM^2)
    motor_front_pwm2 = np.clip(motor_front_pwm2, 0, PWM_MAX_SQ)
    motor_rear_pwm2  = np.clip(motor_rear_pwm2, 0, PWM_MAX_SQ)
    
    # The actual torque generated by saturated motors
    tau_generated = (motor_front_pwm2 - motor_rear_pwm2) / (2.0 * TORQUE_TO_PWM2)
    
    # Actuator lag (Motor response time)
    u_real_pwm2 += dt_sim * (1.0/MOTOR_TAU * (tau_generated - u_real_pwm2))
    
    # ── DISTURBANCE INJECTION ────────────────────────────────
    # At t=8s, a sudden wind gust pushes the nose down (constant bias)
    if disturbance:
        dist_torque = -0.3 if t > 8.0 else 0.0
    else:
        dist_torque = 0.0

    # Plant Physics (Using REAL inertia, not theoretical)
    accel = (u_real_pwm2 + dist_torque) / J_real
    
    x[0,0] += dt_sim * omega
    x[1,0] += dt_sim * accel
    
    # Logging
    t_arr.append(t); x_arr.append([theta, omega])
    cmd_arr.append(theta_cmd); rate_cmd_arr.append(omega_cmd)
    tau_nom_arr.append(tau_nom); tau_ada_arr.append(tau_ada)
    pwm_out_arr.append(np.sqrt(motor_front_pwm2)) # Actual DShot value

# ── 5. PLOT ────────────────────────────────────────────────────────────────
t_arr = np.array(t_arr); x_arr = np.array(x_arr)
cmd_arr = np.array(cmd_arr)
rate_cmd_arr = np.array(rate_cmd_arr)
tau_nom_arr = np.array(tau_nom_arr)
tau_ada_arr = np.array(tau_ada_arr)
pwm_out_arr = np.array(pwm_out_arr)

t_i = 7.0
t_f = 9.0
plot_idx = (t_arr >= t_i) & (t_arr <= t_f)

fig, axes = plt.subplots(4, 1, figsize=(10, 12), sharex=True)

axes[0].plot(t_arr[plot_idx], np.degrees(cmd_arr[plot_idx]), 'k--', label='Command')
axes[0].plot(t_arr[plot_idx], np.degrees(x_arr[plot_idx,0]), 'b-', label='Actual Angle')
axes[0].set_ylabel("Angle [deg]")
axes[0].set_title(f"MRAC: {adaptation} | Real Inertia 30% Higher | Disturbance at T=8s")
axes[0].legend()
axes[0].grid()

axes[1].plot(t_arr[plot_idx], np.degrees(rate_cmd_arr[plot_idx]), 'k--', label='Rate Cmd')
